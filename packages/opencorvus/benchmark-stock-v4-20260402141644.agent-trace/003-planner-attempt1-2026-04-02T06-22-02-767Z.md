- 每 2 秒生成一次新价格：

CREATE TABLE stocks (

  name TEXT NOT NULL,
  open_price REAL NOT NULL,
  current_price REAL NOT NULL,
  high_price REAL NOT NULL,
  low_price REAL NOT NULL,
  volume INTEGER NOT NULL DEFAULT 0,
  trend REAL NOT NULL DEFAULT 0,
  updated_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE price_history (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  price REAL NOT NULL,
  volume INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE orders (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  direction TEXT NOT NULL CHECK(direction IN ('buy', 'sell')),
  quantity INTEGER NOT NULL,
  price REAL NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending' CHECK(status IN ('pending', 'filled', 'cancelled')),
  filled_at INTEGER,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE positions (
  id INTEGER PRIMARY KEY AUTOINCREMENT,

  stock_code TEXT NOT NULL REFERENCES stocks(code),
  quantity INTEGER NOT NULL DEFAULT 0,
  cost_basis REAL NOT NULL DEFAULT 0,
  UNIQUE(user_id, stock_code)
);

CREATE TABLE daily_snapshots (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,

function generateNewPrice(stock: Stock): number {
  // 正态分布随机波动 (σ = 0.5%)
  const randomWalk = (Math.random() + Math.random() + Math.random() + Math.random() - 2) / 2;
  const volatility = 0.005; // 0.5%
  const change = stock.currentPrice * randomWalk * volatility;
  
  // 叠加趋势因子
  const trendChange = stock.currentPrice * stock.trend;
  
  let newPrice = stock.currentPrice + change + trendChange;
  
  // 涨跌停限制 ±10%
  const maxPrice = stock.openPrice * 1.1;
  const minPrice = stock.openPrice * 0.9;
  newPrice = Math.max(minPrice, Math.min(maxPrice, newPrice));
  
  return Math.round(newPrice * 100) / 100; // 保留 2 位小数
}

const API_BASE = '';

async function fetchWithAuth(url: string, options: RequestInit = {}) {
  const token = localStorage.getItem('token');
  const headers = {
    ...options.headers,
    'Content-Type': 'application/json',
    ...(token && { Authorization: `Bearer ${token}` }),
  };
  const res = await fetch(`${API_BASE}${url}`, { ...options, headers });
  if (res.status === 401) {
    localStorage.removeItem('token');
    window.location.href = '/login';
  }
  if (!res.ok) throw new Error(await res.text());
  return res.json();
}

export const apiClient = {
  auth: {
    register: (data) => fetchWithAuth('/api/auth/register', { method: 'POST', body: JSON.stringify(data) }),
    login: (data) => fetchWithAuth('/api/auth/login', { method: 'POST', body: JSON.stringify(data) }),
  },
  market: {
    getStocks: () => fetchWithAuth('/api/market/stocks'),
    getStock: (code) => fetchWithAuth(`/api/market/stocks/${code}`),
  },
  trading: {
    buy: (data) => fetchWithAuth('/api/trading/buy', { method: 'POST', body: JSON.stringify(data) }),
    sell: (data) => fetchWithAuth('/api/trading/sell', { method: 'POST', body: JSON.stringify(data) }),
    cancel: (id) => fetchWithAuth(`/api/trading/cancel/${id}`, { method: 'POST' }),
    getOrders: (params) => fetchWithAuth(`/api/trading/orders?${new URLSearchParams(params)}`),
  },
  portfolio: {
    getPositions: () => fetchWithAuth('/api/portfolio'),
    getSummary: () => fetchWithAuth('/api/portfolio/summary'),
  },
  history: {
    getTrades: (params) => fetchWithAuth(`/api/history/trades?${new URLSearchParams(params)}`),
    getDaily: () => fetchWithAuth('/api/history/daily'),
  },
};

// SSE 订阅
export function subscribeStocks(onData: (stocks) => void, onStatusChange: (connected) => void) {
  const es = new EventSource(`${API_BASE}/api/market/sse`);
  es.onmessage = (e) => onData(JSON.parse(e.data));
  es.onopen = () => onStatusChange(true);
  es.onerror = () => onStatusChange(false);
  return () => es.close();
}

<risks>

1. **SSE 连接管理风险**
   - 风险: 客户端断开连接后服务器未清理资源导致内存泄漏
   - 缓解: 在 Hono SSE 端点中监听 `AbortSignal`，客户端断开时清理 interval

2. **数据库并发风险**
   - 风险: SQLite 在 Bun 单线程中无并发问题，但模拟引擎和交易 API 可能竞争
   - 缓解: 所有数据库操作使用同步 API，利用 JavaScript 单线程特性

3. **价格精度风险**
   - 风险: 浮点数计算导致资金或持仓精度问题
   - 缓解: 所有价格计算使用 `Math.round(value * 100) / 100`，资金计算同样保留 2 位小数

4. **模拟引擎性能风险**
   - 风险: 每 2 秒全量推送可能导致高内存/CPU 使用
   - 缓解: 仅推送股票列表数据，不包含历史价格；历史价格通过单独 API 获取

5. **前端路由刷新风险**
   - 风险: 直接访问 `/stock/SH600001` 等路由时 404
   - 缓解: Hono 静态资源服务配置为 SPA 模式，所有路由返回 index.html

6. **Token 存储安全风险**
   - 风险: localStorage 存储 token 存在 XSS 风险
   - 缓解: 此为 MVP，使用 localStorage 可接受；生产环境应考虑 httpOnly cookie

7. **订单撮合顺序风险**
   - 风险: 价格相同时撮合顺序不明确
   - 缓解: 按 created_at ASC 排序，先挂单的先成交

</risks>

- Q: 模拟引擎撮合时，一个价格变动可能成交多个订单?
  A: 是的，按价格优先、时间优先原则撮合所有可成交订单
