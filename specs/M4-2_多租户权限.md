import type { MiddlewareHandler } from 'hono'
import { AppError } from '../types/errors'

type Permission = 'chat' | 'session:read' | 'session:write' | 'execute' | 'admin'

const ROLE_PERMISSIONS: Record<string, Permission[]> = {
  admin:  ['chat', 'session:read', 'session:write', 'execute', 'admin'],
  user:   ['chat', 'session:read', 'session:write', 'execute'],
  viewer: ['session:read'],
}

export function requirePermission(permission: Permission): MiddlewareHandler {
  return async (c, next) => {
    const userRole = c.get('userRole') as string
    const allowed = ROLE_PERMISSIONS[userRole] || []

    if (!allowed.includes(permission)) {
      throw new AppError('FORBIDDEN', 'Insufficient permissions', 403, { required: permission, role: userRole })
    }

    await next()
  }

## 5. 配额管理

```typescript
// server/services/quota.ts

export class QuotaService {
  async checkQuota(workspaceId: string): Promise<{ allowed: boolean; used: number; limit: number }> {
    const workspace = db.select().from(workspaces).where(eq(workspaces.id, workspaceId)).get()
    if (!workspace) return { allowed: false, used: 0, limit: 0 }

    const today = new Date().toISOString().slice(0, 10)
    const usage = usageRepo.getDailyUsageByWorkspace(workspaceId, today)
    const totalUsed = usage.inputTokens + usage.outputTokens

    return {
      allowed: totalUsed < workspace.dailyTokenQuota,
      used: totalUsed,
      limit: workspace.dailyTokenQuota,
    }
  }
}
```
