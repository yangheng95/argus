## 1. 目标与验收标准

### 1.1 里程碑目标

实现 `IFinanceService.getFinancialReport()`，从 Tushare Pro API 获取 A 股上市公司财报数据（ROE、营收、净利润、毛利率、PE 等），附带 24 小时磁盘缓存。

### 1.2 验收标准

| # | 验收项 | 操作 | 预期结果 |
|---|---|---|---|
| V1 | 单标的查询 | 调用 `getFinancialReport({ tickers: ['600519.SH'], freq: 'quarterly', fields: ['roe'] })` | 返回 FinancialReport[]，≥ 12 条，每条含 `roe` 字段，数值在 5-40 区间 |
| V2 | 多字段查询 | 查询 `['roe','revenue','net_profit','gross_margin','pe_ttm']` | 每条记录 fields 含全部 5 个字段，无 undefined |

## 2. Tushare Pro API 规格

### 2.1 接口协议

- 端点：`https://api.tushare.pro`
- 方法：POST
- Content-Type：`application/json`
- 认证：body 中 `token` 字段

### 2.2 请求格式

```json
{
  "api_name": "fina_indicator",
  "token": "<tushare_token>",
  "params": {
    "ts_code": "600519.SH",
    "start_date": "20210101",
    "end_date": "20241231",
    "fields": "ts_code,ann_date,end_date,roe,grossprofit_margin,netprofit_margin"
  },
  "fields": ""
}
```

### 2.3 响应格式

```json
{
  "request_id": "...",
  "code": 0,
  "msg": null,
  "data": {
    "fields": ["ts_code", "ann_date", "end_date", "roe", "grossprofit_margin"],
    "items": [
      ["600519.SH", "20241030", "20240930", 31.5, 91.2],
      ["600519.SH", "20240430", "20240331", 10.6, 91.8]
    ]
  }
}
```

### 2.4 常用 API 与字段映射

| AimeCode 字段 | Tushare API | Tushare 字段 | 说明 |
|---|---|---|---|
| `roe` | `fina_indicator` | `roe` | 净资产收益率 (%) |
| `revenue` | `income` | `revenue` | 营业收入（元） |
| `net_profit` | `income` | `n_income_attr_p` | 归母净利润（元） |
| `gross_margin` | `fina_indicator` | `grossprofit_margin` | 毛利率 (%) |
| `net_margin` | `fina_indicator` | `netprofit_margin` | 净利率 (%) |
| `pe_ttm` | `daily_basic` | `pe_ttm` | 市盈率 TTM |
| `pb` | `daily_basic` | `pb` | 市净率 |
| `total_revenue_growth` | `fina_indicator` | `or_yoy` | 营收同比增速 (%) |
| `eps` | `fina_indicator` | `eps` | 每股收益（元） |
| `roe_diluted` | `fina_indicator` | `roe_dt` | 摊薄 ROE (%) |

### 2.5 Tushare 日期格式

Tushare 使用 `YYYYMMDD` 格式。AimeCode 使用 `YYYY-MM-DD`。转换在 TushareService 内部完成。

## 3. 完整实现

文件路径：`server/services/tushare-finance.ts`

```typescript
import type { FinancialReport } from '@shared/types'

// ── Tushare 请求类型 ─────────────────────────────────────────
interface TushareRequest {
  api_name: string
  token: string
  params: Record<string, string | number>
  fields?: string
}

interface TushareResponse {
  code: number
  msg: string | null
  data: {
    fields: string[]
    items: (string | number | null)[][]
  } | null
}

// ── 字段映射配置 ─────────────────────────────────────────────
// AimeCode 字段名 → { api: Tushare API 名, field: Tushare 字段名 }
const FIELD_MAP: Record<string, { api: string; field: string }> = {
  roe:                   { api: 'fina_indicator', field: 'roe' },
  roe_diluted:           { api: 'fina_indicator', field: 'roe_dt' },
  gross_margin:          { api: 'fina_indicator', field: 'grossprofit_margin' },
  net_margin:            { api: 'fina_indicator', field: 'netprofit_margin' },
  revenue_growth:        { api: 'fina_indicator', field: 'or_yoy' },
  eps:                   { api: 'fina_indicator', field: 'eps' },
  revenue:               { api: 'income',         field: 'revenue' },
  net_profit:            { api: 'income',         field: 'n_income_attr_p' },
  total_assets:          { api: 'balancesheet',   field: 'total_assets' },
  total_liabilities:     { api: 'balancesheet',   field: 'total_liab' },
  pe_ttm:                { api: 'daily_basic',    field: 'pe_ttm' },
  pb:                    { api: 'daily_basic',    field: 'pb' },
}

// ── 日期工具 ─────────────────────────────────────────────────
function toTushareDate(isoDate: string): string {
  return isoDate.replace(/-/g, '')
}

function toISODate(tushareDate: string): string {
  return `${tushareDate.slice(0, 4)}-${tushareDate.slice(4, 6)}-${tushareDate.slice(6, 8)}`
}

function endDateToPeriod(endDate: string, freq: 'quarterly' | 'annual'): string {
  // endDate 格式: YYYYMMDD（如 20240930）
  const year = endDate.slice(0, 4)
  const month = parseInt(endDate.slice(4, 6), 10)
  if (freq === 'annual') return year
  const quarter = Math.ceil(month / 3)
  return `${year}-Q${quarter}`
}

// ── 缓存 ─────────────────────────────────────────────────────
interface CacheEntry {
  data: FinancialReport[]
  expiresAt: number
}

const cache = new Map<string, CacheEntry>()
const CACHE_TTL_MS = 24 * 60 * 60 * 1000  // 24 小时

function getCacheKey(ticker: string, api: string, fields: string[], freq: string): string {
  return `finance:${ticker}:${api}:${fields.sort().join(',')}:${freq}`
}

// ── TushareFinanceService ────────────────────────────────────
export class TushareFinanceService {
  private token: string
  private baseUrl = 'https://api.tushare.pro'

  constructor(token: string) {
    if (!token || token.trim() === '') {
      throw new Error('Invalid Tushare token: token cannot be empty')
    }
    this.token = token
  }

  async getFinancialReport(params: {
    tickers: string[]
    startDate?: string
    endDate?: string
    freq: 'quarterly' | 'annual'
    fields: string[]
  }): Promise<FinancialReport[]> {
    const results: FinancialReport[] = []

    for (const ticker of params.tickers) {
      const reports = await this.fetchTickerReports(
        ticker,
        params.startDate,
        params.endDate,
        params.freq,
        params.fields,
      )
      results.push(...reports)
    }

    // 按 ticker + period 排序
    results.sort((a, b) => {
      if (a.ticker !== b.ticker) return a.ticker.localeCompare(b.ticker)
      return a.period.localeCompare(b.period)
    })

    return results
  }

  private async fetchTickerReports(
    ticker: string,
    startDate: string | undefined,
    endDate: string | undefined,
    freq: 'quarterly' | 'annual',
    fields: string[],
  ): Promise<FinancialReport[]> {
    // ── 按 Tushare API 分组请求 ──────────────────────────
    const apiFieldGroups = new Map<string, string[]>()
    for (const field of fields) {
      const mapping = FIELD_MAP[field]
      if (!mapping) {
        console.warn(`[TushareFinance] Unknown field: ${field}, skipping`)
        continue
      }
      const group = apiFieldGroups.get(mapping.api) || []
      group.push(field)
      apiFieldGroups.set(mapping.api, group)
    }

    // ── 并行请求各 API ──────────────────────────────────
    const allResults = new Map<string, Map<string, number | null>>()
    // key = period, value = { field: value }

    const promises = Array.from(apiFieldGroups.entries()).map(async ([api, aimecodeFields]) => {
      // 检查缓存
      const cacheKey = getCacheKey(ticker, api, aimecodeFields, freq)
      const cached = cache.get(cacheKey)
      if (cached && Date.now() < cached.expiresAt) {
        console.log(`  [Cache] HIT ${cacheKey}`)
        for (const report of cached.data) {
          const existing = allResults.get(report.period) || new Map()
          for (const [k, v] of Object.entries(report.fields)) {
            existing.set(k, v)
          }
          allResults.set(report.period, existing)
        }
        return
      }

      // 构建 Tushare 请求
      const tushareFields = aimecodeFields
        .map(f => FIELD_MAP[f]!.field)
      const requestFields = ['ts_code', 'end_date', 'ann_date', ...tushareFields].join(',')

      const tushareParams: Record<string, string | number> = {
        ts_code: ticker,
      }
      if (startDate) tushareParams.start_date = toTushareDate(startDate)
      if (endDate) tushareParams.end_date = toTushareDate(endDate)

      // 默认查最近 3 年
      if (!startDate && !endDate) {
        const now = new Date()
        tushareParams.start_date = toTushareDate(
          `${now.getFullYear() - 3}-01-01`
        )
      }

      const response = await this.callTushare(api, tushareParams, requestFields)

      if (!response.data || !response.data.items) {
        console.warn(`[TushareFinance] No data for ${ticker} from ${api}`)
        return
      }

      // 解析响应
      const { fields: respFields, items } = response.data
      const reports: FinancialReport[] = []

      for (const row of items) {
        const rowData: Record<string, any> = {}
        for (let i = 0; i < respFields.length; i++) {
          rowData[respFields[i]] = row[i]
        }

        const endDateRaw = rowData.end_date as string
        if (!endDateRaw) continue

        const period = endDateToPeriod(endDateRaw, freq)
        const existing = allResults.get(period) || new Map()

        for (const aimecodeField of aimecodeFields) {
          const tushareField = FIELD_MAP[aimecodeField]!.field
          const value = rowData[tushareField]
          existing.set(aimecodeField, value !== null && value !== undefined ? Number(value) : null)
        }

        allResults.set(period, existing)
        reports.push({
          ticker,
          period,
          reportDate: toISODate(rowData.ann_date || endDateRaw),
          fields: Object.fromEntries(existing),
        })
      }

      // 写缓存
      cache.set(cacheKey, { data: reports, expiresAt: Date.now() + CACHE_TTL_MS })
      console.log(`  [Cache] SET ${cacheKey} (${reports.length} rows, TTL 24h)`)
    })

    await Promise.all(promises)

    // ── 合并为 FinancialReport[] ─────────────────────────
    const result: FinancialReport[] = []
    for (const [period, fieldsMap] of allResults) {
      result.push({
        ticker,
        period,
        reportDate: '',  // 已在上面赋值
        fields: Object.fromEntries(fieldsMap),
      })
    }

    return result
  }

  private async callTushare(
    apiName: string,
    params: Record<string, string | number>,
    fields: string,
  ): Promise<TushareResponse> {
    const body: TushareRequest = {
      api_name: apiName,
      token: this.token,
      params,
      fields,
    }

    console.log(`  [Tushare] POST ${apiName} ts_code=${params.ts_code}`)

    const response = await fetch(this.baseUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    })

    if (!response.ok) {
      throw new Error(`Tushare API HTTP error: ${response.status}`)
    }

    const json = await response.json() as TushareResponse

    if (json.code !== 0) {
      // Tushare 错误码处理
      if (json.msg?.includes('token') || json.msg?.includes('权限')) {
        throw new Error(`Invalid Tushare token: ${json.msg}`)
      }
      throw new Error(`Tushare API error (code ${json.code}): ${json.msg}`)
    }

    return json
  }
}
```

```typescript
import { TushareFinanceService } from './tushare-finance'
import { MockFinanceService } from './mock-finance'
import type { Context } from 'hono'

export function getFinanceService(c: Context) {
  const authMode = c.get('authMode')

  if (authMode === 'real') {
    const token = c.get('tushareToken')
    if (!token) {
      throw new Error('Tushare token is required for real mode')
    }
    return new TushareFinanceService(token)
  }

  return new MockFinanceService()
}
```

```typescript
import { describe, it, expect, beforeAll } from 'bun:test'
import { TushareFinanceService } from '../tushare-finance'
import type { FinancialReport } from '@shared/types'

// 合约测试：验证返回值结构符合 IFinanceService 契约
// 需要真实 Tushare Token（从环境变量读取）
const TUSHARE_TOKEN = process.env.TUSHARE_TOKEN || ''
const SKIP_REAL = !TUSHARE_TOKEN

describe('TushareFinanceService', () => {
  describe('Constructor', () => {
    it('should throw on empty token', () => {
      expect(() => new TushareFinanceService('')).toThrow('Invalid Tushare token')
    })
  })

  // 以下测试需要真实 Token
  describe.skipIf(SKIP_REAL)('Real API (requires TUSHARE_TOKEN)', () => {

    beforeAll(() => {
      service = new TushareFinanceService(TUSHARE_TOKEN)
    })

    it('should return quarterly ROE for 600519.SH', async () => {
      const reports = await service.getFinancialReport({
        tickers: ['600519.SH'],

        fields: ['roe'],
      })

      expect(Array.isArray(reports)).toBe(true)
      expect(reports.length).toBeGreaterThanOrEqual(4)

      for (const report of reports) {
        expect(report.ticker).toBe('600519.SH')
        expect(report.period).toMatch(/^\d{4}-Q[1-4]$/)

        expect(typeof report.fields.roe).toBe('number')

    })

    it('should return multiple fields', async () => {
      const reports = await service.getFinancialReport({
        tickers: ['600519.SH'],
        freq: 'quarterly',
        fields: ['roe', 'gross_margin', 'net_margin'],
      })

      for (const report of reports) {
        expect(report.fields.roe).toBeDefined()
        expect(report.fields.gross_margin).toBeDefined()
        expect(report.fields.net_margin).toBeDefined()
      }
    })

    it('should return annual data', async () => {
      const reports = await service.getFinancialReport({
        tickers: ['600519.SH'],
        freq: 'annual',
        fields: ['roe'],
      })

      for (const report of reports) {
        expect(report.period).toMatch(/^\d{4}$/)
      }
    })

    it('should support multiple tickers', async () => {
      const reports = await service.getFinancialReport({
        tickers: ['600519.SH', '000858.SZ'],
        freq: 'quarterly',
        fields: ['roe'],
      })

      const tickers = new Set(reports.map(r => r.ticker))
      expect(tickers.has('600519.SH')).toBe(true)
      expect(tickers.has('000858.SZ')).toBe(true)
    })

    it('should cache results', async () => {
      // 首次查询
      const t1 = performance.now()
      await service.getFinancialReport({ tickers: ['600519.SH'], freq: 'quarterly', fields: ['roe'] })
      const d1 = performance.now() - t1

      // 第二次查询（应命中缓存）
      const t2 = performance.now()
      await service.getFinancialReport({ tickers: ['600519.SH'], freq: 'quarterly', fields: ['roe'] })
      const d2 = performance.now() - t2

      expect(d2).toBeLessThan(10) // 缓存应 < 10ms
    })
  })
```

## 6. 集成陷阱

| # | 问题 | 解决方案 |
|---|---|---|
| T1 | Tushare 日期格式 `YYYYMMDD` vs AimeCode `YYYY-MM-DD` | 在 `toTushareDate` / `toISODate` 内部转换 |
| T2 | Tushare 限频（每分钟 200 次） | 缓存 24h + 串行查询同一 ticker（不同 API 可并行） |
| T3 | `fina_indicator` 和 `income` 返回的日期字段不同 | `fina_indicator` 用 `end_date`，`income` 也用 `end_date`；两者用相同的 period 做 key 合并 |
| T4 | PE/PB 来自 `daily_basic`（日频），非季度 | PE/PB 取查询区间最后一个交易日的值，不做季度聚合 |
| T5 | Tushare `code: -2001` 表示 Token 过期或无权限 | 统一映射为 `DATA_SOURCE_ERROR + "Invalid Tushare token"` |
| T6 | 同一字段在不同 API 中名称不同 | 通过 `FIELD_MAP` 统一映射，消费方只用 AimeCode 字段名 |
| T7 | 缓存在进程重启后失效 | M1-2 的 SQLite 可用后，可将缓存落盘；当前阶段用内存缓存 |
| T8 | 并行请求多个 ticker 可能触发限频 | 串行处理 tickers（`for...of` 而非 `Promise.all`），但同一 ticker 的不同 API 可并行 |
