## 2. Docker Compose

文件路径：`docker/docker-compose.yml`

```yaml
version: '3.8'

services:
  # ── API 服务器 ──────────────────────────────────────────
  api:
    build:
      context: ..
      dockerfile: docker/api/Dockerfile
    container_name: aimecode-api
    environment:
      - NODE_ENV=production
      - PORT=3001
      - JWT_SECRET=${JWT_SECRET:?JWT_SECRET is required}
      - ANTHROPIC_API_KEY=${ANTHROPIC_API_KEY:-}
      - TUSHARE_TOKEN=${TUSHARE_TOKEN:-}
      - API_KEY_REQUIRED=${API_KEY_REQUIRED:-false}
      - DATABASE_PATH=/data/aimecode.db
    volumes:
      - aimecode-data:/data
      - /var/run/docker.sock:/var/run/docker.sock  # 用于沙箱容器管理
    ports:
      - "3001:3001"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3001/api/health"]
      interval: 15s
      timeout: 5s
      retries: 3
      start_period: 10s
    restart: unless-stopped
    networks:
      - aimecode

  # ── 前端 ────────────────────────────────────────────────
  web:
    build:
      context: ..
      dockerfile: docker/web/Dockerfile
    container_name: aimecode-web
    depends_on:
      api:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:80/"]
      interval: 15s
      timeout: 5s
      retries: 3
    restart: unless-stopped
    networks:
      - aimecode

  # ── Nginx 反向代理 ──────────────────────────────────────
  nginx:
    image: nginx:1.25-alpine
    container_name: aimecode-nginx
    depends_on:
      web:
        condition: service_healthy
      api:
        condition: service_healthy
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro        # SSL 证书（可选）
    restart: unless-stopped
    networks:
      - aimecode

  # ── 沙箱镜像（预构建）──────────────────────────────────
  sandbox:
    build:
      context: ../docker/sandbox
      dockerfile: Dockerfile
    image: aimecode-sandbox:latest
    # 此服务不运行，仅用于构建镜像
    profiles:
      - build-only

volumes:
  aimecode-data:
    driver: local

networks:
  aimecode:
    driver: bridge
```

## 3. API Dockerfile

文件路径：`docker/api/Dockerfile`

```dockerfile
FROM oven/bun:1.1-alpine

WORKDIR /app

# 复制依赖文件
COPY package.json bun.lockb ./
RUN bun install --production --frozen-lockfile

# 复制源码
COPY server/ ./server/
COPY src/services/types.ts ./src/services/types.ts

# 创建数据目录
RUN mkdir -p /data

EXPOSE 3001

CMD ["bun", "run", "server/index.ts"]
```

## 4. Web Dockerfile

```dockerfile

WORKDIR /app
COPY package.json pnpm-lock.yaml ./
RUN corepack enable && pnpm install --frozen-lockfile

RUN pnpm build

FROM nginx:1.25-alpine
COPY --from=builder /app/dist /usr/share/nginx/html
COPY docker/web/nginx-spa.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
```

## 5. Nginx 配置

文件路径：`docker/nginx/nginx.conf`

```nginx
events { worker_connections 1024; }

http {
  upstream api { server api:3001; }
  upstream web { server web:80; }

  server {
    listen 80;
    listen 443 ssl;
    server_name _;

    ssl_certificate /etc/nginx/ssl/cert.pem;
    ssl_certificate_key /etc/nginx/ssl/key.pem;

    # API 路由
    location /api/ {
      proxy_pass http://api;
      proxy_http_version 1.1;
      proxy_set_header Upgrade $http_upgrade;
      proxy_set_header Connection "upgrade";
      proxy_set_header Host $host;
      proxy_set_header X-Real-IP $remote_addr;
      proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

      # SSE 支持
      proxy_buffering off;
      proxy_cache off;
      proxy_read_timeout 300s;
    }

    # 前端
    location / {
      proxy_pass http://web;
    }
  }
}
```

## 7. 备份恢复脚本

```bash
#!/bin/bash
# docker/scripts/backup.sh

BACKUP_DIR="./backups"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="${BACKUP_DIR}/aimecode-${DATE}.tar.gz"

mkdir -p "$BACKUP_DIR"

# 停止 API 以确保数据一致性
docker compose stop api

# 备份 volume 数据
docker run --rm \
  -v aimecode-data:/data:ro \
  -v "$(pwd)/${BACKUP_DIR}":/backup \
  alpine tar czf "/backup/aimecode-${DATE}.tar.gz" -C /data .

# 重启
docker compose start api

echo "Backup saved to ${BACKUP_FILE}"
```

## 8. Air-gap 离线部署

```bash
# 在有网络的机器上：
docker compose build
docker save aimecode-api aimecode-web aimecode-sandbox nginx:1.25-alpine | gzip > aimecode-images.tar.gz

# 复制到离线机器：
docker load < aimecode-images.tar.gz
docker compose up -d
```

## 9. 集成陷阱

| # | 问题 | 解决方案 |
|---|---|---|
| T1 | Docker socket 挂载是安全风险 | 生产环境改用 Docker TCP + TLS 或 rootless Docker |
| T2 | SQLite 在 volume 中 WAL 文件需要同一 volume | `aimecode.db`, `.db-wal`, `.db-shm` 在同一 `/data` 目录 |
| T3 | Nginx SSL 证书不存在时启动失败 | 提供自签名证书生成脚本；或只用 HTTP 模式 |
| T4 | SSE 被 Nginx 缓冲 | `proxy_buffering off; proxy_cache off;` |
| T5 | Air-gap 环境无法 pull base image | `docker save` 导出所有依赖镜像 |
