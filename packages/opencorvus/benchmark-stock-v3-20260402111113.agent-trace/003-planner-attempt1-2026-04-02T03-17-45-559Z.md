CREATE TABLE users (

CREATE TABLE stocks (
  code TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  open_price REAL NOT NULL,
  current_price REAL NOT NULL,
  high_price REAL NOT NULL,
  low_price REAL NOT NULL,
  volume INTEGER NOT NULL DEFAULT 0,
  trend REAL NOT NULL DEFAULT 0,
  updated_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE price_history (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  price REAL NOT NULL,
  volume INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE orders (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  direction TEXT NOT NULL CHECK(direction IN ('buy', 'sell')),
  quantity INTEGER NOT NULL,
  price REAL NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending' CHECK(status IN ('pending', 'filled', 'cancelled')),
  filled_at INTEGER,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE positions (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  quantity INTEGER NOT NULL DEFAULT 0,
  cost_basis REAL NOT NULL DEFAULT 0,
  UNIQUE(user_id, stock_code)
);

CREATE TABLE daily_snapshots (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,

<risks>
- 风险: 并行目标会共同依赖 `index.ts` 的导出契约，若 Goal 1 与 Goal 6 约定不一致，Goal 7 的测试导入会失效。
  mitigation: 从一开始固定 `buildApp(options): { app, db, simulator }` 与 `export const app = buildApp().app` 签名，后续只补实现不改接口。

- 风险: PRD 原始表结构没有显式“冻结资金/冻结持仓/成交价”字段，若直接硬做会让撤单恢复与历史成交价格难以精确实现。
  mitigation: 在 `src/db.ts` 的 `orders` 表最小补充 `reserved_cash`、`reserved_quantity`、`fill_price` 三个字段，并在计划内统一让交易、撮合、历史复用。

- 风险: SSE 在 Bun/Hono 下若未正确处理客户端断开，长时间运行会累积失效连接。
  mitigation: `src/market.ts` 维护客户端集合，连接关闭/写入失败时立即移除；增加心跳注释避免中间代理超时。

- 风险: 成交测试如果依赖真实 2 秒定时器，`bun test` 会慢且不稳定。
  mitigation: `src/simulator.ts` 必须提供 `tickOnce()`；`buildApp({ enableSimulator: false })` 下仍可返回可手动推进的 simulator 控制器给测试使用。

- 风险: 卖出冻结语义若直接扣减 `positions.quantity`，持仓页与可卖数量会混淆。
  mitigation: 卖单只记录 `reserved_quantity` 在 `orders`，持仓总量保持真实已持有数量；可卖数量通过 `positions.quantity - pending sell sum(reserved_quantity)` 计算。

- 风险: `daily_snapshots` 只在成交时写入，刚注册的新用户可能没有任何历史曲线数据。
  mitigation: `GET /api/history/daily` 在无记录时回退生成当天一条当前总资产点，保证前端图表可渲染。

- 风险: 前端 React Router v7 与具体包名/API 可能因版本差异有细节变化。
  mitigation: 以 `web/package.json` 实际安装的 Router v7 包为准；若使用 `react-router-dom` 的现代数据路由 API，保持页面文件命名和路由路径不变即可，不要在根计划外引入其它路由方案。
</risks>
