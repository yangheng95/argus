# AimeCode M4-4 · 宏观数据工具

> **自包含文档**：一个 AI 模型只读这一个文件，就能实现宏观经济指标查询服务。
> **前置条件**：M1-3（TushareService 基础架构）+ M1-1（API 框架）已完成。
> **包含**：GDP/CPI/PMI/利率/汇率/社融等宏观指标查询、IFinanceService 扩展、Tushare 宏观 API 对接、LLM 工具定义、Mock 数据。
> **不包含**：实时利率推送、宏观预测模型、Wind 宏观数据（M4-6）。

export interface MacroIndicator {
  date: string                // 'YYYY-MM-DD' 或 'YYYY-QN'

  unit: string                // '%', '点', '亿元'

  endDate: string
  freq?: 'daily' | 'monthly' | 'quarterly'

## 4. Tushare 宏观数据服务

```typescript
// server/services/tushare-macro.ts

const MACRO_API_MAP: Record<string, { api: string; field: string; freq: string; unit: string }> = {
  gdp_yoy:   { api: 'cn_gdp',   field: 'gdp_yoy', freq: 'quarterly', unit: '%' },
  cpi_yoy:   { api: 'cn_cpi',   field: 'nt_yoy',  freq: 'monthly',   unit: '%' },
  ppi_yoy:   { api: 'cn_ppi',   field: 'ppi_yoy', freq: 'monthly',   unit: '%' },
  pmi_mfg:   { api: 'cn_pmi',   field: 'pmi',     freq: 'monthly',   unit: '点' },
  m2_yoy:    { api: 'cn_m',     field: 'm2_yoy',  freq: 'monthly',   unit: '%' },
  shibor_on: { api: 'shibor',   field: 'on',      freq: 'daily',     unit: '%' },
  shibor_1w: { api: 'shibor',   field: '1w',      freq: 'daily',     unit: '%' },
  shibor_1m: { api: 'shibor',   field: '1m',      freq: 'daily',     unit: '%' },
  lpr_1y:    { api: 'cn_lpr',   field: 'lpr_1y',  freq: 'monthly',   unit: '%' },
  lpr_5y:    { api: 'cn_lpr',   field: 'lpr_5y',  freq: 'monthly',   unit: '%' },
}

export class TushareMacroService {
  private token: string

  constructor(token: string) {
    this.token = token
  }

  async getMacroData(params: MacroDataParams): Promise<MacroIndicator[]> {
    const results: MacroIndicator[] = []

    // 按 API 分组
    const apiGroups = new Map<string, string[]>()
    for (const indicator of params.indicators) {
      const mapping = MACRO_API_MAP[indicator]
      if (!mapping) continue
      const group = apiGroups.get(mapping.api) || []
      group.push(indicator)
      apiGroups.set(mapping.api, group)
    }

    // 并行请求
    await Promise.all(Array.from(apiGroups.entries()).map(async ([api, indicators]) => {
      const fields = indicators.map(i => MACRO_API_MAP[i].field)
      const response = await this.callTushare(api, {
        start_date: params.startDate.replace(/-/g, ''),
        end_date: params.endDate.replace(/-/g, ''),
      }, ['month', 'date', ...fields].join(','))

      if (!response.data?.items) return

      const { fields: respFields, items } = response.data
      for (const row of items) {
        const rowData: Record<string, any> = {}
        for (let i = 0; i < respFields.length; i++) {
          rowData[respFields[i]] = row[i]
        }

        const dateRaw = rowData.date || rowData.month
        if (!dateRaw) continue

        for (const indicator of indicators) {
          const mapping = MACRO_API_MAP[indicator]
          const value = rowData[mapping.field]
          results.push({
            date: formatDate(dateRaw),
            indicator,
            value: value !== null && value !== undefined ? Number(value) : null,
            unit: mapping.unit,
          })
        }
      }
    }))

    return results.sort((a, b) => a.date.localeCompare(b.date))
  }

  private async callTushare(apiName: string, params: Record<string, string>, fields: string) {
    const res = await fetch('https://api.tushare.pro', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ api_name: apiName, token: this.token, params, fields }),
    })
    return res.json()
  }
}

function formatDate(raw: string): string {
  if (raw.length === 6) return `${raw.slice(0, 4)}-${raw.slice(4, 6)}`  // YYYYMM → YYYY-MM
  if (raw.length === 8) return `${raw.slice(0, 4)}-${raw.slice(4, 6)}-${raw.slice(6, 8)}`
  return raw
}
```

## 6. 集成陷阱

| # | 问题 | 解决方案 |
|---|---|---|
| T1 | Tushare 宏观 API 需要较高积分权限 | 用户需 2000+ 积分；Mock 数据兜底开发 |
| T2 | GDP 按季度发布，CPI 按月 | `freq` 参数让用户或 LLM 指定，或自动推断 |
| T3 | SHIBOR 日度数据量大 | 缓存 24h + 只查最近 1 年 |
| T4 | 日期字段格式不一致 | `formatDate` 统一处理 YYYYMM 和 YYYYMMDD |
