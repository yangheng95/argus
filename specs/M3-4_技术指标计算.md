# AimeCode M3-4 · 技术指标计算

> **自包含文档**：一个 AI 模型只读这一个文件，就能实现服务端技术指标计算服务。
> **前置条件**：M1-4（行情数据服务，OHLCVBar 数据可获取）已完成。
> **包含**：MA / EMA / MACD / RSI / Bollinger Bands / KDJ / ATR / OBV 的纯 TypeScript 实现、API 端点、图表 spec 生成。
> **不包含**：自定义指标公式编辑器、实时计算、分钟级数据。

## 1. 目标与验收标准

### 1.1 里程碑目标

在服务端提供常用技术指标的计算能力，LLM 可通过工具调用获取计算结果，或在生成的 Python 代码中引用。

### 1.2 验收标准

| # | 验收项 | 操作 | 预期结果 |
|---|---|---|---|
| V1 | MA 计算 | `calcMA(bars, [5, 10, 20])` | 每个 bar 附加 ma5/ma10/ma20 字段，前 N-1 个为 null |
| V2 | MACD 计算 | `calcMACD(bars)` | 每个 bar 含 dif/dea/macd_hist 字段 |
| V3 | RSI 计算 | `calcRSI(bars, 14)` | 值在 0-100 之间 |
| V4 | Bollinger | `calcBollinger(bars, 20, 2)` | 含 upper/middle/lower，middle = MA20 |
| V5 | 图表生成 | 茅台日线 + MACD subplot | 返回 Plotly spec：上图 K线+MA，下图 MACD 柱状图 |
| V6 | API 端点 | `POST /api/indicators` | 传入 ticker + 日期范围 + 指标列表，返回计算结果 |
| V7 | 数值精度 | 与 TradingView 对比 | MA/RSI 误差 < 0.01% |

## 2. 技术指标实现

文件路径：`server/services/indicators/index.ts`

```typescript
import type { OHLCVBar } from '@shared/types'

export interface IndicatorResult extends OHLCVBar {
  [key: string]: number | string | null | undefined
}

// ── MA (简单移动平均) ────────────────────────────────────────
export function calcMA(bars: OHLCVBar[], periods: number[]): IndicatorResult[] {
  const result: IndicatorResult[] = bars.map(b => ({ ...b }))

  for (const period of periods) {
    const key = `ma${period}`
    for (let i = 0; i < result.length; i++) {
      if (i < period - 1) {
        result[i][key] = null
      } else {
        let sum = 0
        for (let j = i - period + 1; j <= i; j++) {
          sum += bars[j].close
        }
        result[i][key] = Math.round(sum / period * 100) / 100
      }
    }
  }

  return result
}

// ── EMA (指数移动平均) ──────────────────────────────────────
export function calcEMA(values: number[], period: number): (number | null)[] {
  const result: (number | null)[] = new Array(values.length).fill(null)
  const multiplier = 2 / (period + 1)

  // 前 period-1 个为 null，第 period 个用 SMA 初始化
  let sum = 0
  for (let i = 0; i < period; i++) {
    sum += values[i]
  }
  result[period - 1] = sum / period

  for (let i = period; i < values.length; i++) {
    result[i] = values[i] * multiplier + (result[i - 1] as number) * (1 - multiplier)
  }

  return result
}

// ── MACD ─────────────────────────────────────────────────────
export function calcMACD(
  bars: OHLCVBar[],
  fastPeriod = 12,
  slowPeriod = 26,
  signalPeriod = 9,
): IndicatorResult[] {
  const closes = bars.map(b => b.close)
  const emaFast = calcEMA(closes, fastPeriod)
  const emaSlow = calcEMA(closes, slowPeriod)

  // DIF = EMA(fast) - EMA(slow)
  const dif: (number | null)[] = closes.map((_, i) => {
    if (emaFast[i] === null || emaSlow[i] === null) return null
    return Math.round(((emaFast[i] as number) - (emaSlow[i] as number)) * 100) / 100
  })

  // DEA = EMA(DIF, signal)
  const difValues = dif.filter((v): v is number => v !== null)
  const deaRaw = calcEMA(difValues, signalPeriod)

  // 对齐长度
  const deaOffset = dif.findIndex(v => v !== null)
  const dea: (number | null)[] = new Array(bars.length).fill(null)
  for (let i = 0; i < deaRaw.length; i++) {
    dea[deaOffset + i] = deaRaw[i] !== null ? Math.round(deaRaw[i]! * 100) / 100 : null
  }

  return bars.map((b, i) => ({
    ...b,
    dif: dif[i],
    dea: dea[i],
    macd_hist: dif[i] !== null && dea[i] !== null
      ? Math.round(((dif[i] as number) - (dea[i] as number)) * 2 * 100) / 100
      : null,
  }))
}

// ── RSI ──────────────────────────────────────────────────────
export function calcRSI(bars: OHLCVBar[], period = 14): IndicatorResult[] {
  const result: IndicatorResult[] = bars.map(b => ({ ...b, rsi: null as number | null }))

  if (bars.length < period + 1) return result

  // 计算涨跌幅
  const changes: number[] = []
  for (let i = 1; i < bars.length; i++) {
    changes.push(bars[i].close - bars[i - 1].close)
  }

  // 首个 RSI：用 SMA 计算 avg gain / avg loss
  let avgGain = 0
  let avgLoss = 0
  for (let i = 0; i < period; i++) {
    if (changes[i] > 0) avgGain += changes[i]
    else avgLoss += Math.abs(changes[i])
  }
  avgGain /= period
  avgLoss /= period

  result[period].rsi = avgLoss === 0 ? 100 : Math.round(100 - 100 / (1 + avgGain / avgLoss) * 100) / 100

  // 后续 RSI：用 Wilder 平滑
  for (let i = period; i < changes.length; i++) {
    const gain = changes[i] > 0 ? changes[i] : 0
    const loss = changes[i] < 0 ? Math.abs(changes[i]) : 0

    avgGain = (avgGain * (period - 1) + gain) / period
    avgLoss = (avgLoss * (period - 1) + loss) / period

    result[i + 1].rsi = avgLoss === 0 ? 100 : Math.round((100 - 100 / (1 + avgGain / avgLoss)) * 100) / 100
  }

  return result
}

// ── Bollinger Bands ──────────────────────────────────────────
export function calcBollinger(
  bars: OHLCVBar[],
  period = 20,
  stdDev = 2,
): IndicatorResult[] {
  const result: IndicatorResult[] = bars.map(b => ({
    ...b,
    bb_upper: null as number | null,
    bb_middle: null as number | null,
    bb_lower: null as number | null,
  }))

  for (let i = period - 1; i < bars.length; i++) {
    let sum = 0
    for (let j = i - period + 1; j <= i; j++) sum += bars[j].close
    const mean = sum / period

    let variance = 0
    for (let j = i - period + 1; j <= i; j++) {
      variance += (bars[j].close - mean) ** 2
    }
    const std = Math.sqrt(variance / period)

    result[i].bb_middle = Math.round(mean * 100) / 100
    result[i].bb_upper = Math.round((mean + stdDev * std) * 100) / 100
    result[i].bb_lower = Math.round((mean - stdDev * std) * 100) / 100
  }

  return result
}

// ── KDJ ──────────────────────────────────────────────────────
export function calcKDJ(bars: OHLCVBar[], period = 9): IndicatorResult[] {
  const result: IndicatorResult[] = bars.map(b => ({
    ...b,
    k: null as number | null,
    d: null as number | null,
    j: null as number | null,
  }))

  let prevK = 50
  let prevD = 50

  for (let i = period - 1; i < bars.length; i++) {
    let highest = -Infinity
    let lowest = Infinity
    for (let j = i - period + 1; j <= i; j++) {
      highest = Math.max(highest, bars[j].high)
      lowest = Math.min(lowest, bars[j].low)
    }

    const rsv = highest === lowest ? 50 : ((bars[i].close - lowest) / (highest - lowest)) * 100
    const k = 2 / 3 * prevK + 1 / 3 * rsv
    const d = 2 / 3 * prevD + 1 / 3 * k
    const j = 3 * k - 2 * d

    result[i].k = Math.round(k * 100) / 100
    result[i].d = Math.round(d * 100) / 100
    result[i].j = Math.round(j * 100) / 100

    prevK = k
    prevD = d
  }

  return result
}

// ── ATR (平均真实波幅) ──────────────────────────────────────
export function calcATR(bars: OHLCVBar[], period = 14): IndicatorResult[] {
  const result: IndicatorResult[] = bars.map(b => ({ ...b, atr: null as number | null }))

  if (bars.length < 2) return result

  // True Range
  const tr: number[] = [bars[0].high - bars[0].low]
  for (let i = 1; i < bars.length; i++) {
    tr.push(Math.max(
      bars[i].high - bars[i].low,
      Math.abs(bars[i].high - bars[i - 1].close),
      Math.abs(bars[i].low - bars[i - 1].close),
    ))
  }

  // ATR = SMA(TR) 然后 Wilder 平滑
  let atr = 0
  for (let i = 0; i < period; i++) atr += tr[i]
  atr /= period
  result[period - 1].atr = Math.round(atr * 100) / 100

  for (let i = period; i < bars.length; i++) {
    atr = (atr * (period - 1) + tr[i]) / period
    result[i].atr = Math.round(atr * 100) / 100
  }

  return result
}
```

```typescript
import type { IndicatorResult } from './index'

/**
 * 生成带技术指标的 Plotly chart spec。
 * 上图：K 线 + MA 线
 * 下图：MACD 柱状图 + DIF/DEA 线
 */
export function buildKLineWithMACD(
  data: IndicatorResult[],
  title: string,
): object {
  return {
    data: [
      // K 线
      {
        type: 'candlestick',
        x: data.map(d => d.date),
        open: data.map(d => d.open),
        high: data.map(d => d.high),
        low: data.map(d => d.low),
        close: data.map(d => d.close),
        increasing: { line: { color: '#10B981' }, fillcolor: 'rgba(16,185,129,0.3)' },
        decreasing: { line: { color: '#EF4444' }, fillcolor: 'rgba(239,68,68,0.3)' },
        name: 'K线',
        xaxis: 'x',
        yaxis: 'y',
      },
      // MA 线
      ...[
        { key: 'ma5', color: '#E8A838', name: 'MA5' },
        { key: 'ma10', color: '#3B82F6', name: 'MA10' },
        { key: 'ma20', color: '#A855F7', name: 'MA20' },
      ].filter(m => data.some(d => d[m.key] !== null && d[m.key] !== undefined)).map(m => ({
        type: 'scatter',
        mode: 'lines',
        x: data.map(d => d.date),
        y: data.map(d => d[m.key]),
        line: { color: m.color, width: 1 },
        name: m.name,
        xaxis: 'x',
        yaxis: 'y',
      })),
      // MACD 柱状图
      ...(data.some(d => d.macd_hist != null) ? [
        {
          type: 'bar',
          x: data.map(d => d.date),
          y: data.map(d => d.macd_hist),
          marker: {
            color: data.map(d => (d.macd_hist as number) >= 0 ? '#10B981' : '#EF4444'),
          },
          name: 'MACD',
          xaxis: 'x2',
          yaxis: 'y2',
        },
        {
          type: 'scatter',
          mode: 'lines',
          x: data.map(d => d.date),
          y: data.map(d => d.dif),
          line: { color: '#3B82F6', width: 1 },
          name: 'DIF',
          xaxis: 'x2',
          yaxis: 'y2',
        },
        {
          type: 'scatter',
          mode: 'lines',
          x: data.map(d => d.date),
          y: data.map(d => d.dea),
          line: { color: '#F59E0B', width: 1 },
          name: 'DEA',
          xaxis: 'x2',
          yaxis: 'y2',
        },
      ] : []),
    ],
    layout: {
      title: { text: title, font: { size: 14 } },
      grid: { rows: 2, columns: 1, pattern: 'independent', roworder: 'top to bottom' },
      xaxis: { rangeslider: { visible: false } },
      yaxis: { title: '价格', side: 'right', domain: [0.3, 1] },
      xaxis2: { matches: 'x' },
      yaxis2: { title: 'MACD', side: 'right', domain: [0, 0.25] },
      showlegend: true,
      legend: { orientation: 'h', y: 1.02 },
    },
  }
}
```

## 5. 集成陷阱

| # | 问题 | 解决方案 |
|---|---|---|
| T1 | EMA 初始值影响精度 | 用 SMA 初始化 EMA 首个值 |
| T2 | MACD 柱状图在不同软件中定义不同 | 统一用 `(DIF - DEA) * 2` |
| T3 | RSI 有 SMA 和 Wilder 两种算法 | 使用 Wilder 平滑（与大多数金融软件一致） |
| T4 | 数据不足时前 N 个值为 null | 客户端渲染需过滤 null 点 |
| T5 | KDJ 的 K/D 初始值 | 初始 K=50, D=50（业界惯例） |
