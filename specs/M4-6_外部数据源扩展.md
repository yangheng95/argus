# AimeCode M4-6 · 外部数据源扩展

> **自包含文档**：一个 AI 模型只读这一个文件，就能实现多数据源适配器框架（Wind / AkShare / Bloomberg）。
> **前置条件**：M1-3（TushareService 已实现 IFinanceService 接口）+ M1-1（服务注册 registry）已完成。
> **包含**：数据源适配器抽象、Wind API 适配器、AkShare 适配器、数据源自动切换、配置管理、SourceTag 显示。
> **不包含**：Bloomberg 终端集成（需要专线）、实时 tick 数据。

---

## 1. 目标与验收标准

| # | 验收项 | 操作 | 预期结果 |
|---|---|---|---|
| V1 | AkShare 适配 | 配置 `DATA_SOURCE=akshare` | 财报/行情查询改走 AkShare HTTP API |
| V2 | Wind 适配 | 配置 `DATA_SOURCE=wind` + Wind API key | 财报/行情查询改走 Wind |
| V3 | SourceTag | 查询返回后检查 UI | ToolCallBadge 显示 "来源：Wind" 或 "来源：AkShare" |
| V4 | 自动切换 | Wind API 连接失败 | 日志记录失败，配置切换到下一个源（非 fallback，需手动配置优先级） |
| V5 | 合约测试通用 | AkShare 通过同一套合约测试 | 返回结构与 Tushare 完全一致 |

## 3. AkShare 适配器

```typescript
// server/services/finance/akshare-adapter.ts

import type { IFinanceAdapter } from './adapter'
import type { FinancialReport, OHLCVBar } from '@shared/types'

/**
 * AkShare 适配器。
 * AkShare 是开源免费的 Python 金融数据接口，通过 HTTP 代理调用。
 * 需要运行 AkShare HTTP 服务：`pip install akshare && aktools --host 0.0.0.0 --port 8080`
 */
export class AkShareAdapter implements IFinanceAdapter {
  readonly name = 'akshare'
  readonly displayName = 'AkShare'
  private baseUrl: string

  constructor(baseUrl: string = 'http://localhost:8080') {
    this.baseUrl = baseUrl
  }

  async getFinancialReport(params: {
    tickers: string[]
    freq: 'quarterly' | 'annual'
    fields: string[]
  }): Promise<FinancialReport[]> {
    const results: FinancialReport[] = []

    for (const ticker of params.tickers) {
      // AkShare 接口：stock_financial_analysis_indicator
      const response = await fetch(`${this.baseUrl}/api/public/stock_financial_analysis_indicator`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ symbol: ticker.replace(/\.(SH|SZ)$/, '') }),
      })

      const data = await response.json()
      // 转换为 FinancialReport 格式
      // ... 字段映射逻辑 ...
    }

    return results
  }

  async getOHLCV(params: {
    ticker: string
    startDate: string
    endDate: string
    freq?: 'D' | 'W' | 'M'
  }): Promise<OHLCVBar[]> {
    const symbol = params.ticker.replace(/\.(SH|SZ)$/, '')
    const response = await fetch(`${this.baseUrl}/api/public/stock_zh_a_hist`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        symbol,
        period: params.freq === 'W' ? 'weekly' : params.freq === 'M' ? 'monthly' : 'daily',
        start_date: params.startDate.replace(/-/g, ''),
        end_date: params.endDate.replace(/-/g, ''),
      }),
    })

    const data = await response.json()
    // 转换为 OHLCVBar[]
    return (data || []).map((row: any) => ({
      date: row['日期'],
      open: Number(row['开盘']),
      high: Number(row['最高']),
      low: Number(row['最低']),
      close: Number(row['收盘']),
      volume: Number(row['成交量']),
      amount: Number(row['成交额']),
      turnover: Number(row['换手率']),
      change_pct: Number(row['涨跌幅']),
    }))
  }

  async getIndexComponents(params: { indexCode: string }) {
    // AkShare: index_stock_cons
    const response = await fetch(`${this.baseUrl}/api/public/index_stock_cons`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ symbol: params.indexCode.replace(/\.(SH|SZ)$/, '') }),
    })
    const data = await response.json()
    return (data || []).map((row: any) => ({
      ticker: row['品种代码'] + (row['品种代码'].startsWith('6') ? '.SH' : '.SZ'),
      name: row['品种名称'] || '',
      weight: 0,  // AkShare 不返回权重
    }))
  }

  async testConnection(): Promise<boolean> {
    try {
      const res = await fetch(`${this.baseUrl}/api/public`)
      return res.ok
    } catch {
      return false
    }
  }
}
```

## 4. Wind 适配器骨架

```typescript
// server/services/finance/wind-adapter.ts

import type { IFinanceAdapter } from './adapter'

/**
 * Wind 数据源适配器。
 * 通过 Wind 量化接口（WindPy HTTP Proxy）调用。
 * 需要 Wind 客户端 + WindPy 代理服务。
 */
export class WindAdapter implements IFinanceAdapter {
  readonly name = 'wind'
  readonly displayName = 'Wind'
  private baseUrl: string
  private apiKey: string

  constructor(baseUrl: string, apiKey: string) {
    this.baseUrl = baseUrl
    this.apiKey = apiKey
  }

  async getFinancialReport(params: any) {
    // Wind: w.wsd("600519.SH", "roe,grossmargin", "2021-01-01", "2024-12-31", "rptType=1")
    const response = await fetch(`${this.baseUrl}/wsd`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`,
      },
      body: JSON.stringify({
        codes: params.tickers.join(','),
        fields: this.mapFields(params.fields),
        startDate: params.startDate,
        endDate: params.endDate,
        options: `rptType=${params.freq === 'annual' ? '1' : '0'}`,
      }),
    })

    const data = await response.json()
    // 转换为 FinancialReport[]
    return this.parseWindFinancial(data, params.tickers, params.fields)
  }

  async getOHLCV(params: any) {
    // Wind: w.wsd("600519.SH", "open,high,low,close,volume,amt", "2024-01-01", "2024-12-31")
    const response = await fetch(`${this.baseUrl}/wsd`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`,
      },
      body: JSON.stringify({
        codes: params.ticker,
        fields: 'open,high,low,close,volume,amt,turn,pct_chg',
        startDate: params.startDate,
        endDate: params.endDate,
        options: `Period=${params.freq || 'D'}`,
      }),
    })

    const data = await response.json()
    return this.parseWindOHLCV(data)
  }

  async getIndexComponents(params: any) {
    // Wind: w.wset("sectorconstituent", "date=2024-01-01;sectorid=000300.SH")
    const response = await fetch(`${this.baseUrl}/wset`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.apiKey}`,
      },
      body: JSON.stringify({
        tableName: 'sectorconstituent',
        options: `date=${params.date || new Date().toISOString().slice(0, 10)};sectorid=${params.indexCode}`,
      }),
    })

    const data = await response.json()
    return this.parseWindComponents(data)
  }

  async testConnection(): Promise<boolean> {
    try {
      const res = await fetch(`${this.baseUrl}/health`, {
        headers: { 'Authorization': `Bearer ${this.apiKey}` },
      })
      return res.ok
    } catch {
      return false
    }
  }

  private mapFields(fields: string[]): string {
    const WIND_FIELD_MAP: Record<string, string> = {
      roe: 'roe_ttm2', gross_margin: 'grossprofitmargin',
      net_margin: 'netprofitmargin', pe_ttm: 'pe_ttm',
      pb: 'pb_lf', revenue: 'or_ttm', net_profit: 'netprofit_ttm2',
    }
    return fields.map(f => WIND_FIELD_MAP[f] || f).join(',')
  }

  private parseWindFinancial(data: any, tickers: string[], fields: string[]) { return [] }
  private parseWindOHLCV(data: any) { return [] }
  private parseWindComponents(data: any) { return [] }
}
```

## 5. 数据源注册与切换

```typescript
// server/services/finance/registry.ts

import type { IFinanceAdapter } from './adapter'
import { TushareFinanceService } from '../tushare-finance'
import { TushareMarketService } from '../tushare-market'
import { AkShareAdapter } from './akshare-adapter'
import { WindAdapter } from './wind-adapter'

type DataSourceName = 'tushare' | 'akshare' | 'wind'

const adapters = new Map<DataSourceName, IFinanceAdapter>()

export function registerAdapter(name: DataSourceName, adapter: IFinanceAdapter) {
  adapters.set(name, adapter)
}

export function getAdapter(preference?: DataSourceName): IFinanceAdapter {
  const name = preference || (process.env.DATA_SOURCE as DataSourceName) || 'tushare'
  const adapter = adapters.get(name)
  if (!adapter) throw new Error(`Data source '${name}' not registered`)
  return adapter
}

// 初始化时注册可用的适配器
export function initAdapters() {
  const tushareToken = process.env.TUSHARE_TOKEN
  if (tushareToken) {
    // Tushare 不直接实现 IFinanceAdapter，需要封装
    registerAdapter('tushare', createTushareAdapter(tushareToken))
  }

  const akshareUrl = process.env.AKSHARE_URL
  if (akshareUrl) {
    registerAdapter('akshare', new AkShareAdapter(akshareUrl))
  }

  const windUrl = process.env.WIND_URL
  const windKey = process.env.WIND_API_KEY
  if (windUrl && windKey) {
    registerAdapter('wind', new WindAdapter(windUrl, windKey))
  }
}
```

## 7. 集成陷阱

| # | 问题 | 解决方案 |
|---|---|---|
| T1 | AkShare HTTP 代理不稳定 | 配置超时 + 重试；AkShare 更适合开发测试而非生产 |
| T2 | Wind 需要本地客户端 | Wind 适配器通过 HTTP Proxy 调用，需要部署 WindPy 代理服务 |
| T3 | 不同数据源的字段名不同 | 每个适配器内部做字段映射，对外统一为 AimeCode 字段名 |
| T4 | AkShare 返回中文列名 | 在 adapter 中做中文→英文列名映射 |
| T5 | 数据源切换不是 fallback | 通过配置 `DATA_SOURCE` 显式选择，不自动降级 |
| T6 | AkShare 不返回指数权重 | weight 设为 0，前端判断 weight=0 时不显示权重列 |
