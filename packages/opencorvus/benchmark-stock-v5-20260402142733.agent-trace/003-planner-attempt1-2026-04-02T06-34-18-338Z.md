CREATE TABLE users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  username TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  balance REAL NOT NULL DEFAULT 1000000,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE sessions (
  token TEXT PRIMARY KEY,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE stocks (
  code TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  open_price REAL NOT NULL,
  current_price REAL NOT NULL,
  high_price REAL NOT NULL,
  low_price REAL NOT NULL,
  volume INTEGER NOT NULL DEFAULT 0,
  trend REAL NOT NULL DEFAULT 0,
  updated_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE price_history (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  price REAL NOT NULL,
  volume INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE orders (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  direction TEXT NOT NULL CHECK(direction IN ('buy', 'sell')),
  quantity INTEGER NOT NULL,
  price REAL NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending' CHECK(status IN ('pending', 'filled', 'cancelled')),
  filled_at INTEGER,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

CREATE TABLE positions (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  stock_code TEXT NOT NULL REFERENCES stocks(code),
  quantity INTEGER NOT NULL DEFAULT 0,
  cost_basis REAL NOT NULL DEFAULT 0,
  UNIQUE(user_id, stock_code)
);

CREATE TABLE daily_snapshots (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  total_assets REAL NOT NULL,
  created_at INTEGER NOT NULL DEFAULT (unixepoch())
);

import { Hono } from "hono";

export const authRouter = new Hono();

// Middleware to extract and validate Bearer token
export async function authMiddleware(c: any, next: any) {
  const authHeader = c.req.header("Authorization");
  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return c.json({ error: "Unauthorized" }, 401);
  }
  
  const token = authHeader.slice(7);
  const session = db.query("SELECT user_id FROM sessions WHERE token = ?").get(token) as { user_id: number } | undefined;
  
  if (!session) {
    return c.json({ error: "Invalid token" }, 401);
  }
  
  c.set("userId", session.user_id);
  await next();
}

// POST /api/auth/register
authRouter.post("/register", async (c) => {
  const { username, password } = await c.req.json();
  
  if (!username || !password || username.length < 3 || password.length < 6) {
    return c.json({ error: "Invalid username or password" }, 400);
  }
  
  const passwordHash = await Bun.password.hash(password, { algorithm: "argon2id" });
  
  try {
    const result = db.query(
      "INSERT INTO users (username, password_hash) VALUES (?, ?) RETURNING id, username, balance, created_at"
    ).get(username, passwordHash) as { id: number; username: string; balance: number; created_at: number };
    
    return c.json({
      id: result.id,
      username: result.username,
      balance: result.balance,
      created_at: result.created_at,
    }, 201);
  } catch (e) {
    return c.json({ error: "Username already exists" }, 409);
  }
});

// POST /api/auth/login
authRouter.post("/login", async (c) => {
  const { username, password } = await c.req.json();

import { db } from "./db.js";

// Box-Muller transform for normal distribution
function randomNormal(): number {
  let u = 0, v = 0;
  while (u === 0) u = Math.random();
  while (v === 0) v = Math.random();
  return Math.sqrt(-2.0 * Math.log(u)) * Math.cos(2.0 * Math.PI * v);
}

export function generatePrice(currentPrice: number, trend: number, openPrice: number): number {
  // Random walk with σ=0.5%
  const volatility = 0.005;
  const randomChange = randomNormal() * volatility;
  
  // Apply trend and random change
  let newPrice = currentPrice * (1 + randomChange + trend);
  
  // Enforce ±10% limit from open price
  const maxPrice = openPrice * 1.1;
  const minPrice = openPrice * 0.9;
  newPrice = Math.max(minPrice, Math.min(maxPrice, newPrice));
  
  // Round to 2 decimal places
  return Math.round(newPrice * 100) / 100;
}

export function matchOrders(stockCode: string, currentPrice: number) {
  // Match buy orders: market price <= buy price
  const buyOrders = db.query(`
    SELECT o.id, o.user_id, o.quantity, o.price
    FROM orders o
    WHERE o.stock_code = ? AND o.direction = 'buy' AND o.status = 'pending' AND o.price >= ?
    ORDER BY o.price DESC, o.created_at ASC
  `).all(stockCode, currentPrice) as Array<{ id: number; user_id: number; quantity: number; price: number }>;
  
  for (const order of buyOrders) {
    fillBuyOrder(order, currentPrice);
  }
  
  // Match sell orders: market price >= sell price
  const sellOrders = db.query(`
    SELECT o.id, o.user_id, o.quantity, o.price
    FROM orders o
    WHERE o.stock_code = ? AND o.direction = 'sell' AND o.status = 'pending' AND o.price <= ?
    ORDER BY o.price ASC, o.created_at ASC
  `).all(stockCode, currentPrice) as Array<{ id: number; user_id: number; quantity: number; price: number }>;
  
  for (const order of sellOrders) {
    fillSellOrder(order, currentPrice);
  }
}

function fillBuyOrder(order: { id: number; user_id: number; quantity: number; price: number }, fillPrice: number) {
  // Update order status
  db.query("UPDATE orders SET status = 'filled', filled_at = unixepoch() WHERE id = ?").run(order.id);
  
  // Update or create position
  const position = db.query("SELECT id, quantity, cost_basis FROM positions WHERE user_id = ? AND stock_code = ?")
    .get(order.user_id, order.stock_code) as { id: number; quantity: number; cost_basis: number } | undefined;
  
  if (position) {
    const totalCost = position.quantity * position.cost_basis + order.quantity * fillPrice;
    const newQuantity = position.quantity + order.quantity;
    const newCostBasis = totalCost / newQuantity;
    db.query("UPDATE positions SET quantity = ?, cost_basis = ? WHERE id = ?")
      .run(newQuantity, newCostBasis, position.id);
  } else {
    db.query("INSERT INTO positions (user_id, stock_code, quantity, cost_basis) VALUES (?, ?, ?, ?)")
      .run(order.user_id, order.stock_code, order.quantity, fillPrice);
  }
}

function fillSellOrder(order: { id: number; user_id: number; quantity: number; price: number }, fillPrice: number) {
  // Update order status
  db.query("UPDATE orders SET status = 'filled', filled_at = unixepoch() WHERE id = ?").run(order.id);
  
  // Update position
  const position = db.query("SELECT id, quantity FROM positions WHERE user_id = ? AND stock_code = ?")
    .get(order.user_id, order.stock_code) as { id: number; quantity: number } | undefined;
  
  if (position) {
    const newQuantity = position.quantity - order.quantity;
    if (newQuantity <= 0) {
      db.query("DELETE FROM positions WHERE id = ?").run(position.id);
    } else {
      db.query("UPDATE positions SET quantity = ? WHERE id = ?").run(newQuantity, position.id);
    }
  }
  
  // Return funds to user (sell proceeds)
  const proceeds = order.quantity * fillPrice;
  db.query("UPDATE users SET balance = balance + ? WHERE id = ?").run(proceeds, order.user_id);
}

export function startSimulator() {
  // Generate initial 60 price points on startup
  const stocks = db.query("SELECT code, open_price, trend FROM stocks").all() as 
    Array<{ code: string; open_price: number; trend: number }>;
  
  for (const stock of stocks) {
    let currentPrice = stock.open_price;
    const now = Math.floor(Date.now() / 1000);
    
    for (let i = 60; i > 0; i--) {
      const timestamp = now - i * 2; // 2 seconds apart
      currentPrice = generatePrice(currentPrice, stock.trend, stock.open_price);
      db.query("INSERT INTO price_history (stock_code, price, created_at) VALUES (?, ?, ?)")
        .run(stock.code, currentPrice, timestamp);
    }
    
    // Update current price in stocks table
    db.query("UPDATE stocks SET current_price = ?, high_price = ?, low_price = ? WHERE code = ?")
      .run(currentPrice, currentPrice, currentPrice, stock.code);
  }
  
  // Start price update interval
  setInterval(() => {
    const stocks = db.query("SELECT code, current_price, open_price, trend FROM stocks").all() as 
      Array<{ code: string; current_price: number; open_price: number; trend: number }>;
    
    for (const stock of stocks) {
      const newPrice = generatePrice(stock.current_price, stock.trend, stock.open_price);
      
      // Update high/low
      const { high_price, low_price } = db.query("SELECT high_price, low_price FROM stocks WHERE code = ?")
        .get(stock.code) as { high_price: number; low_price: number };
      
      const newHigh = Math.max(high_price, newPrice);
      const newLow = Math.min(low_price, newPrice);
      
      db.query("UPDATE stocks SET current_price = ?, high_price = ?, low_price = ?, updated_at = unixepoch() WHERE code = ?")
        .run(newPrice, newHigh, newLow, stock.code);
      
      // Record price history
      db.query("INSERT INTO price_history (stock_code, price) VALUES (?, ?)").run(stock.code, newPrice);
      
      // Match orders
      matchOrders(stock.code, newPrice);
    }
  }, 2000);
}

**File**: `src/history.ts`

**Implementation**:
```typescript
import { Hono } from "hono";
import { db } from "./db.js";
import { authMiddleware } from "./auth.js";

export const historyRouter = new Hono();
historyRouter.use("*", authMiddleware);

// GET /api/history/trades
historyRouter.get("/trades", (c) => {
  const userId = c.get("userId");
  const stock = c.req.query("stock");
  const limit = parseInt(c.req.query("limit") || "50");
  const offset = parseInt(c.req.query("offset") || "0");
  
  let query = `
    SELECT 
      o.id,
      o.stock_code,
      s.name as stock_name,
      o.direction,
      o.quantity,
      o.price,
      ROUND(o.quantity * o.price, 2) as amount,
      o.filled_at as trade_time
    FROM orders o
    JOIN stocks s ON o.stock_code = s.code
    WHERE o.user_id = ? AND o.status = 'filled'
  `;
  const params: (number | string)[] = [userId];
  
  if (stock) {
    query += " AND o.stock_code = ?";
    params.push(stock);
  }
  
  query += " ORDER BY o.filled_at DESC LIMIT ? OFFSET ?";
  params.push(limit, offset);
  
  const trades = db.query(query).all(...params);
  
  // Get total count for pagination
  let countQuery = "SELECT COUNT(*) as count FROM orders WHERE user_id = ? AND status = 'filled'";
  const countParams: (number | string)[] = [userId];
  if (stock) {
    countQuery += " AND stock_code = ?";
    countParams.push(stock);
  }
  const { count } = db.query(countQuery).get(...countParams) as { count: number };
  
  return c.json({
    trades,
    pagination: { total: count, limit, offset },
  });
});

// GET /api/history/daily
historyRouter.get("/daily", (c) => {
  const userId = c.get("userId");
  
  const snapshots = db.query(`
    SELECT 
      date(created_at, 'unixepoch') as date,
      total_assets
    FROM daily_snapshots
    WHERE user_id = ?
    ORDER BY created_at DESC
    LIMIT 30
  `).all(userId);
  
  return c.json(snapshots.reverse());
});
```

### 8. Main Entry (`index.ts`)

**File**: `index.ts`

**Implementation**:
```typescript
import { Hono } from "hono";
import { serveStatic } from "hono/bun";
import { initDatabase } from "./src/db.js";
import { authRouter } from "./src/auth.js";
import { marketRouter, setupSSE } from "./src/market.js";
import { tradingRouter } from "./src/trading.js";
import { portfolioRouter } from "./src/portfolio.js";
import { historyRouter } from "./src/history.js";
import { startSimulator } from "./src/simulator.js";

// Initialize database
initDatabase();

const app = new Hono();

// API routes
app.route("/api/auth", authRouter);
app.route("/api/market", marketRouter);
app.route("/api/trading", tradingRouter);
app.route("/api/portfolio", portfolioRouter);
app.route("/api/history", historyRouter);

// SSE endpoint
setupSSE(app);

// Serve static files
app.use("/*", serveStatic({ root: "./web/dist" }));

// SPA fallback
app.get("*", serveStatic({ path: "./web/dist/index.html" }));

// Start simulator
startSimulator();

console.log("Server starting on http://localhost:3000");
export default {
  port: 3000,
  fetch: app.fetch,
};
```

async function resetDatabase() {
  const db = new Database(join(import.meta.dir, "..", "data", "trading.db"));
  db.exec("DELETE FROM sessions");
  db.exec("DELETE FROM orders");
  db.exec("DELETE FROM positions");
  db.exec("DELETE FROM daily_snapshots");
  db.exec("DELETE FROM users WHERE id > 0");
  db.exec("DELETE FROM price_history");
  // Reset stocks to initial state
  db.exec("UPDATE stocks SET current_price = open_price, high_price = open_price, low_price = open_price, volume = 0");
  db.close();
}

describe("Stock Trading API", () => {
