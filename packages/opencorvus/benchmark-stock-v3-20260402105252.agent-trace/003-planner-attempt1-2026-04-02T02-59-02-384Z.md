  cost_basis REAL NOT NULL DEFAULT 0,
  UNIQUE(user_id, stock_code)
);

CREATE TABLE daily_snapshots (

<risks>
- 风险: PRD 表结构未显式包含冻结资金/冻结持仓字段，但目标 4 要求冻结语义一致。
  mitigation: 在 `src/db.ts` 将 `users.frozen_balance` 与 `positions.frozen_quantity` 作为实现性扩展列加入幂等迁移；API 对外仍遵循原 PRD 字段，不影响验收。

- 风险: EventSource 无法方便携带 Bearer token，若将 SSE 设计为私有接口会增加前端复杂度。
  mitigation: 将 `/api/market/sse` 保持公开，只推送公共行情快照；私有数据仍走 Bearer fetch 接口。

- 风险: 测试若依赖真实 2 秒定时器会导致不稳定和耗时。
  mitigation: 在 `src/simulator.ts` 暴露 `tick()`，并让 `index.ts` 支持 `startSimulator: false`；测试手动触发撮合。

- 风险: 重复启动服务可能导致 `price_history` 和 `daily_snapshots` 无限制增长，破坏“60 个点”和“30 天曲线”的稳定性。
  mitigation: `initDb()` 对种子历史采用“今日不足才补齐”，对每日快照采用“每天每用户 upsert 一条”策略。

- 风险: 前后端字段命名若混用 snake_case/camelCase，会导致页面集成期大量返工。
  mitigation: 从后端 JSON 响应开始统一用 camelCase；数据库层独立保留 snake_case。

- 风险: 根级 TS 配置较简，若 web 与后端共用配置可能造成 Vite/DOM/Bun 类型冲突。
  mitigation: 保持 `web/tsconfig.json` 独立；根级 `tsconfig.json` 只覆盖后端与测试，必要时通过 `include/exclude` 明确边界。
</risks>
